-- dws层-----------------------------------------------------------------------------------
--0:各国家总电影
drop table test_dws.gj_m_sum;

create  table test_dws.gj_m_sum
(
    gjname string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.gj_m_sum
SELECT
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END AS a,
    count(*) AS c
FROM test_dwd.movies m
GROUP BY
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END;


--0.1各国家好评电影数
drop table test_dws.gj_m_h_sum;

create  table test_dws.gj_m_h_sum
(
    gjname string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.gj_m_h_sum
SELECT
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END AS a,
    count(*) AS c
FROM test_dwd.movies m
where m.bf>7.5
GROUP BY
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END;



-- 1：根据时间分类，统计每年各国家电影有多少部
drop table test_dws.lx_g_t_sum;

create  table test_dws.lx_g_t_sum
(
    gjname string,
    ltiam string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.lx_g_t_sum
SELECT
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END AS a,
    trim(substr(sj, 0, 4)) AS b,
    count(*) AS c
FROM test_dwd.movies m
GROUP BY
    CASE
        WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
        ELSE split(gj, '/')[0]
        END,
    trim(substr(sj, 0, 4));



-- 2：根据国家分类，统计各国家各种类的电影有多少部
drop table test_dws.lx_g_sum;

create  table test_dws.lx_g_sum
(
    gjname string,
    ltiam string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.lx_g_sum
SELECT
    CASE
       WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
       ELSE split(gj, '/')[0]
       END AS a,
    trim(split(lx, '/')[0]) leixin,
    count(*) num
from test_dwd.movies
GROUP BY
    trim(split(lx, '/')[0]),
     CASE
       WHEN split(gj, '/')[0] LIKE '%中国%' THEN '中国'
       ELSE split(gj, '/')[0]
    END;

-- 3：统计各类型电影的评论数
drop table test_dws.lx_m_c_sum;

create  table test_dws.lx_m_c_sum
(
    lxname string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.lx_m_c_sum
select
    trim(split(lx, '/')[0]) ,
count(A.pl)
FROM test_dwd.comments A
         JOIN test_dwd.movies B
                   ON A.mid=B.id
group by trim(split(lx, '/')[0]);

--3.1 各类型电影的好评数
drop table test_dws.lx_m_c_h_sum;

create  table test_dws.lx_m_c_h_sum
(
    lxname string,
    sum    int
)row format delimited
    fields terminated by ',';

insert into table test_dws.lx_m_c_h_sum
select
    trim(split(lx, '/')[0]) ,
    count(A.pl)
FROM test_dwd.comments A
         JOIN test_dwd.movies B
              ON A.mid=B.id
where A.pf>3
group by trim(split(lx, '/')[0]);

-- ads层操作-----------------------------------------------------------------------------------
-- 1用户对各国家电影的好评率
drop table test_ads.gj_m_hpl;

create  table test_ads.gj_m_hpl
(
    gj string,
    m_sum    int,
    m_h_sum    int,
    m_h_lu    string
)row format delimited
    fields terminated by ',';

insert into table test_ads.gj_m_hpl
SELECT
    m.gjname,
    m.sum,
    COALESCE(hrm.sum, 0) AS highly_rated_movies,
    ROUND(if(m.sum > 0, (COALESCE(hrm.sum, 0) / m.sum) * 100, 0), 2) AS rating_rate
FROM test_dws.gj_m_sum m
         LEFT JOIN test_dws.gj_m_h_sum hrm ON m.gjname = hrm.gjname;
